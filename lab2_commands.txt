docker compose up -d — запуск усіх сервісів у фоновому режимі.

docker compose down — зупинка та видалення контейнерів, мереж, створених командою up.

docker compose build --no-cache — «чиста» збірка контейнерів без використання кешу (допомогло застосувати оновлення Dockerfile).

docker compose logs -f api — перегляд логів контейнера api у реальному часі (дуже корисно для налагодження).

docker ps — перевірка списку запущених контейнерів.

docker compose exec api poetry add sqlalchemy psycopg2-binary — встановлення необхідних бібліотек безпосередньо всередину вже працюючого контейнера.
